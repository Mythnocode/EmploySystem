void insertAllPersonData() {
	printf("请输入需要的人数\n");
	int index = 1;
	int n=0;
	int inednum = 0;
	scanf("%d", &n);
	getchar();
	int all_num = n * 6;
	FILE* fp = fopen("data.txt", "a");
	char *temp=(char*)malloc((all_num*20)*sizeof(char));
	memset(temp, 0, all_num * 20);
	char line[100];
	printf("请输入需要的数据\n");
	while (fgets(line, sizeof(line), stdin) != NULL&&inednum<n) {
		line[strcspn(line, "\n")] = '\0';
		strcat(temp, line);
		strcat(temp, " ");
		inednum++;
	}
	inednum = 0;
	char* token = strtok(temp, " ");
	while (token != NULL&&inednum<=all_num) {
		fprintf(fp, "%s", token);
		if (index % 6 == 0) {
			fprintf(fp, "\n");
		}
		else {
			fprintf(fp, " ");
		}
		index++;
		inednum++;
		token = strtok(NULL, " ");
	}
	fclose(fp);
	printf("写入成功！\n");
	system("pause");
}
